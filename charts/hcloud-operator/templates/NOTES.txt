  kubectl -n {{ .Release.Namespace }} get pods -l app.kubernetes.io/instance={{ .Release.Name }}

Then create your first Hetzner resource:

  cat <<YAML | kubectl apply -f -
  apiVersion: hcloud.shebanglabs.io/v1alpha1
  kind: HetznerServer
  metadata:
    name: web-01
  spec:
    serverType: cpx11
    image: ubuntu-24.04
    location: nbg1
  YAML

  kubectl get hetznerservers -w
